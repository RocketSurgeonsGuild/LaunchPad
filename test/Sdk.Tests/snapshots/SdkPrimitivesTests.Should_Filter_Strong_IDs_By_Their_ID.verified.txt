schema {
  query: Query
}

type Query {
  rows(where: RowFilterInput): [Row!]!
}

type Row {
  id: ID!
  name: String!
}

input IdOperationFilterInput {
  eq: ID
  neq: ID
  in: [ID]
  nin: [ID]
}

input RowFilterInput {
  and: [RowFilterInput!]
  or: [RowFilterInput!]
  id: IdOperationFilterInput
  name: StringOperationFilterInput
}

input StringOperationFilterInput {
  and: [StringOperationFilterInput!]
  or: [StringOperationFilterInput!]
  eq: String
  neq: String
  contains: String
  ncontains: String
  in: [String]
  nin: [String]
  startsWith: String
  nstartsWith: String
  endsWith: String
  nendsWith: String
}
